\begin{frame}{Problem 1}
Prove directly from the definition that
\[
  \frac{1}{n}\longrightarrow 0.
\]
\end{frame}

\begin{frame}{Solution 1}
\begin{proof}
Let $\epsilon>0$. Choose
\[
  N=\left\lfloor\frac{1}{\epsilon}\right\rfloor+1,
  \qquad\text{so that }N>\frac{1}{\epsilon}.
\]
For every $n\geq N$,
\[
  \left\lvert\frac{1}{n}-0\right\rvert
  =\frac{1}{n}\leq\frac{1}{N}<\epsilon.
\]
Since $\epsilon>0$ was arbitrary, $1/n\to0$.
\end{proof}
\end{frame}

\begin{frame}{Problem 2}
Prove directly from the definition that
\[
  \frac{3n+1}{n+2}\longrightarrow 3.
\]
\end{frame}

\begin{frame}{Solution 2}
\begin{proof}
Let $\epsilon>0$. Choose
\[
  N=\left\lfloor\frac{5}{\epsilon}\right\rfloor+1,
  \qquad\text{so that }N>\frac{5}{\epsilon}.
\]
For every $n\geq N$,
\[
\begin{aligned}
  \left\lvert\frac{3n+1}{n+2}-3\right\rvert
  &=\left\lvert\frac{-5}{n+2}\right\rvert\\
  &=\frac{5}{n+2}\leq\frac{5}{n}
    \leq\frac{5}{N}<\epsilon.
\end{aligned}
\]
Thus $(3n+1)/(n+2)\to3$.
\end{proof}
\end{frame}

\begin{frame}{Problem 3}
Prove directly from the definition that
\[
  \frac{2n^2-1}{n^2+3}\longrightarrow 2.
\]
\end{frame}

\begin{frame}{Solution 3}
\begin{proof}
Let $\epsilon>0$. Choose
\[
  N=\left\lfloor\sqrt{\frac{7}{\epsilon}}\right\rfloor+1,
  \qquad\text{so that }N>\sqrt{\frac{7}{\epsilon}}.
\]
For every $n\geq N$,
\[
\begin{aligned}
  \left\lvert\frac{2n^2-1}{n^2+3}-2\right\rvert
  &=\frac{7}{n^2+3}\\
  &\leq\frac{7}{n^2}\leq\frac{7}{N^2}<\epsilon.
\end{aligned}
\]
Thus $(2n^2-1)/(n^2+3)\to2$.
\end{proof}
\end{frame}

\begin{frame}{Problem 4}
Prove directly from the definition that
\[
  \frac{(-1)^n}{n}\longrightarrow 0.
\]
\end{frame}

\begin{frame}{Solution 4}
\begin{proof}
Let $\epsilon>0$. Choose
\[
  N=\left\lfloor\frac{1}{\epsilon}\right\rfloor+1.
\]
For every $n\geq N$,
\[
  \left\lvert\frac{(-1)^n}{n}-0\right\rvert
  =\frac{\lvert(-1)^n\rvert}{n}
  =\frac{1}{n}\leq\frac{1}{N}<\epsilon.
\]
Thus $(-1)^n/n\to0$.
\end{proof}
\end{frame}

\begin{frame}{Problem 5}
Prove directly from the definition that
\[
  \left(1+\frac{1}{n}\right)^2\longrightarrow 1.
\]
\textbf{Hint:} Expand the square and use $1/n^2\leq1/n$.
\end{frame}

\begin{frame}{Solution 5}
\begin{proof}
Let $\epsilon>0$. Choose
\[
  N=\left\lfloor\frac{3}{\epsilon}\right\rfloor+1.
\]
For every $n\geq N$,
\[
\begin{aligned}
  \left\lvert\left(1+\frac{1}{n}\right)^2-1\right\rvert
  &=\frac{2}{n}+\frac{1}{n^2}\\
  &\leq\frac{3}{n}\leq\frac{3}{N}<\epsilon.
\end{aligned}
\]
Thus $(1+1/n)^2\to1$.
\end{proof}
\end{frame}

\begin{frame}{Problem 6}
Prove directly from the definition that
\[
  \sqrt{n^2+1}-n\longrightarrow 0.
\]
\textbf{Hint:} Rationalize the expression.
\end{frame}

\begin{frame}{Solution 6}
\begin{proof}
Let $\epsilon>0$. Choose
\[
  N=\left\lfloor\frac{1}{2\epsilon}\right\rfloor+1.
\]
For every $n\geq N$,
\[
\begin{aligned}
  \left\lvert\sqrt{n^2+1}-n\right\rvert
  &=\frac{(n^2+1)-n^2}{\sqrt{n^2+1}+n}\\
  &\leq\frac{1}{2n}\leq\frac{1}{2N}<\epsilon.
\end{aligned}
\]
Thus $\sqrt{n^2+1}-n\to0$.
\end{proof}
\end{frame}

\begin{frame}{Problem 7}
Prove directly from the definition that
\[
  \frac{\sin n}{\sqrt{n}}\longrightarrow 0.
\]
You may use $\lvert\sin t\rvert\leq1$ for every real number $t$.
\end{frame}

\begin{frame}{Solution 7}
\begin{proof}
Let $\epsilon>0$. Choose
\[
  N=\left\lfloor\frac{1}{\epsilon^2}\right\rfloor+1,
  \qquad\text{so that }\frac{1}{\sqrt{N}}<\epsilon.
\]
For every $n\geq N$,
\[
  \left\lvert\frac{\sin n}{\sqrt{n}}-0\right\rvert
  =\frac{\lvert\sin n\rvert}{\sqrt{n}}
  \leq\frac{1}{\sqrt{n}}\leq\frac{1}{\sqrt{N}}<\epsilon.
\]
Thus $\sin n/\sqrt{n}\to0$.
\end{proof}
\end{frame}

\begin{frame}{Problem 8}
Prove directly from the definition that
\[
  \frac{1}{2^n}\longrightarrow 0.
\]
You may use $2^n\geq n+1$ for every $n\geq1$.
\medskip

\textbf{Optional:} Prove this inequality by induction.
\end{frame}

\begin{frame}{Solution 8}
\begin{proof}
Let $\epsilon>0$. Choose
\[
  N=\left\lfloor\frac{1}{\epsilon}\right\rfloor+1.
\]
Using $2^n\geq n+1$, for every $n\geq N$ we obtain
\[
  \left\lvert\frac{1}{2^n}-0\right\rvert
  \leq\frac{1}{n+1}<\frac{1}{n}\leq\frac{1}{N}<\epsilon.
\]
Thus $1/2^n\to0$.
\end{proof}
\medskip

For the optional induction, $2^1=2$. If $2^k\geq k+1$, then
\[
  2^{k+1}\geq2(k+1)\geq k+2.
\]
\end{frame}

\begin{frame}{Problem 9}
Suppose a real sequence satisfies
\[
  \lvert a_n-2\rvert\leq\frac{3}{n}
  \qquad(n\geq1).
\]
Prove directly from the definition that
\[
  a_n\longrightarrow2
  \qquad\text{and}\qquad
  a_n^2\longrightarrow4.
\]
\end{frame}

\begin{frame}{Solution 9}
\begin{proof}
Let $\epsilon>0$. For $n\geq1$, the triangle inequality gives
\[
  \lvert a_n+2\rvert\leq\lvert a_n-2\rvert+4
  \leq\frac{3}{n}+4\leq7.
\]
Choose $N=\lfloor21/\epsilon\rfloor+1$. For every $n\geq N$,
\[
\begin{aligned}
  \lvert a_n-2\rvert
    &\leq\frac{3}{n}\leq\frac{3}{N}<\epsilon,\\
  \lvert a_n^2-4\rvert
    &=\lvert a_n-2\rvert\lvert a_n+2\rvert
      \leq\frac{21}{n}\leq\frac{21}{N}<\epsilon.
\end{aligned}
\]
Thus $a_n\to2$ and $a_n^2\to4$.
\end{proof}
\end{frame}

\begin{frame}{Problem 10}
Prove directly from the definition that
\[
  a_n=(-1)^n
\]
does not converge in $\mathbb R$ with the Euclidean metric.
\medskip

For every proposed limit $L$, find $\epsilon>0$ such that for
every positive integer $N$, some $n\geq N$ satisfies
\[
  \lvert a_n-L\rvert\geq\epsilon.
\]
\end{frame}

\begin{frame}{Solution 10}
\begin{proof}
Let $L\in\mathbb R$ be any proposed limit. Take $\epsilon=1$.
Let $N$ be any positive integer.
\medskip

If $L\leq0$, choose the even index $n=2N$. Then $n\geq N$ and
\[
  \lvert a_n-L\rvert=\lvert1-L\rvert=1-L\geq1.
\]
If $L>0$, choose the odd index $n=2N+1$. Then $n\geq N$ and
\[
  \lvert a_n-L\rvert=\lvert-1-L\rvert=1+L>1.
\]
Thus no $N$ works for $\epsilon=1$. Since $L$ was arbitrary,
the sequence does not converge.
\end{proof}
\end{frame}

\subsubsection{Sequences in the plane}

\begin{frame}{Problem 11}
Using Euclidean distance in $\mathbb R^2$, prove that
\[
  \left(\frac{1}{n},\frac{2}{n}\right)\longrightarrow(0,0).
\]
Recall that
\[
  d\bigl((x,y),(a,b)\bigr)
  =\sqrt{(x-a)^2+(y-b)^2}.
\]
\end{frame}

\begin{frame}{Solution 11}
\begin{proof}
Let $\epsilon>0$. Choose
\[
  N=\left\lfloor\frac{\sqrt{5}}{\epsilon}\right\rfloor+1.
\]
For every $n\geq N$,
\[
\begin{aligned}
  d\left(\left(\frac{1}{n},\frac{2}{n}\right),(0,0)\right)
  &=\sqrt{\frac{1}{n^2}+\frac{4}{n^2}}\\
  &=\frac{\sqrt{5}}{n}\leq\frac{\sqrt{5}}{N}<\epsilon.
\end{aligned}
\]
Thus $(1/n,2/n)\to(0,0)$.
\end{proof}
\end{frame}

\begin{frame}{Problem 12}
Using Euclidean distance in $\mathbb R^2$, prove that
\[
  \left(\frac{n}{n+1},\frac{(-1)^n}{\sqrt{n}}\right)
  \longrightarrow(1,0).
\]
\textbf{Hint:} Make each coordinate error less than $\epsilon/2$.
\end{frame}

\begin{frame}{Solution 12}
\begin{proof}
Let $\epsilon>0$. Choose
$N=\lfloor\max\{2/\epsilon,4/\epsilon^2\}\rfloor+1$.
For every $n\geq N$,
\[
\begin{aligned}
  \left\lvert\frac{n}{n+1}-1\right\rvert
  &=\frac{1}{n+1}\leq\frac{1}{N}<\frac{\epsilon}{2},\\
  \left\lvert\frac{(-1)^n}{\sqrt{n}}\right\rvert
  &=\frac{1}{\sqrt{n}}\leq\frac{1}{\sqrt{N}}<\frac{\epsilon}{2}.
\end{aligned}
\]
The Euclidean distance to $(1,0)$ therefore satisfies
\[
  \sqrt{\frac{1}{(n+1)^2}+\frac{1}{n}}
  \leq\frac{1}{n+1}+\frac{1}{\sqrt{n}}<\epsilon.
\]
Thus the sequence converges to $(1,0)$.
\end{proof}
\end{frame}

\begin{frame}{Problem 13}
Using Euclidean distance in $\mathbb R^2$, prove that
\[
  \left(\frac{\cos n}{n},\frac{\sin n}{n}\right)
  \longrightarrow(0,0).
\]
You may use $\cos^2t+\sin^2t=1$.
\end{frame}

\begin{frame}{Solution 13}
\begin{proof}
Let $\epsilon>0$. Choose
\[
  N=\left\lfloor\frac{1}{\epsilon}\right\rfloor+1.
\]
For every $n\geq N$,
\[
\begin{aligned}
  d\left(\left(\frac{\cos n}{n},\frac{\sin n}{n}\right),(0,0)\right)
  &=\sqrt{\frac{\cos^2 n+\sin^2 n}{n^2}}\\
  &=\frac{1}{n}\leq\frac{1}{N}<\epsilon.
\end{aligned}
\]
Thus $(\cos n/n,\sin n/n)\to(0,0)$.
\end{proof}
\end{frame}

\subsubsection{Sequences in metric spaces}

\begin{frame}{Problem 14}
Equip $\mathbb R$ with the metric
\[
  d(x,y)=\sqrt{\lvert x-y\rvert}.
\]
Prove directly that $1/n\to0$ in this metric.
\medskip

Compare your choice of $N(\epsilon)$ with your choice for the
Euclidean metric in Problem 1.
\end{frame}

\begin{frame}{Solution 14}
\begin{proof}
Let $\epsilon>0$. Choose
\[
  N=\left\lfloor\frac{1}{\epsilon^2}\right\rfloor+1.
\]
For every $n\geq N$,
\[
  d\left(\frac{1}{n},0\right)
  =\sqrt{\left\lvert\frac{1}{n}-0\right\rvert}
  =\frac{1}{\sqrt{n}}\leq\frac{1}{\sqrt{N}}<\epsilon.
\]
Thus $1/n\to0$ in this metric.
\end{proof}
\medskip

The Euclidean proof used $N>1/\epsilon$. This proof uses
$N>1/\epsilon^2$ because $d(1/n,0)<\epsilon$ requires
$1/n<\epsilon^2$.
\end{frame}

\begin{frame}{Problem 15}
Equip a set $X$ with the discrete metric
\[
  d(x,y)=
  \begin{cases}
    0,&x=y,\\
    1,&x\neq y.
  \end{cases}
\]
Prove that a sequence $(x_n)$ converges to $p\in X$ if and only
if it is \textbf{eventually equal to $p$}.
\medskip

That is, there exists a positive integer $N_0$ such that
$x_n=p$ for every $n\geq N_0$.
\end{frame}

\begin{frame}{Solution 15}
\begin{proof}
Suppose $x_n\to p$. Take $\epsilon=1$. By the definition of
convergence, there is a positive integer $N_0$ such that
\[
  n\geq N_0\quad\Longrightarrow\quad d(x_n,p)<1.
\]
The discrete metric takes only the values $0$ and $1$, so
$d(x_n,p)<1$ forces $x_n=p$. Thus the sequence is eventually
equal to $p$.
\medskip

Conversely, suppose $x_n=p$ for every $n\geq N_0$.
Let $\epsilon>0$. Choose $N=N_0$. Then for every $n\geq N$,
\[
  d(x_n,p)=0<\epsilon.
\]
Thus $x_n\to p$.
\end{proof}
\end{frame}

\begin{frame}{Problem 16}
Prove directly from the definition that
\[
  \frac{3^n}{n!}\longrightarrow0.
\]
\textbf{Hint:} If $a_n=3^n/n!$, show that
\[
  a_{n+1}\leq\frac{a_n}{2}\qquad(n\geq5).
\]
Then compare with the geometric estimate in Problem 8.
\end{frame}

\begin{frame}{Solution 16}
\begin{proof}
Let $\epsilon>0$. Choose $N=6+\lfloor2/\epsilon\rfloor$,
so that $N\geq6$ and $N-5>2/\epsilon$.
\medskip

For $n\geq N$, each factor $3/k$ with $k\geq7$ is at most $1/2$.
Since $3^6/6!=81/80<2$,
\[
\begin{aligned}
  \left\lvert\frac{3^n}{n!}-0\right\rvert
  &=\frac{3^6}{6!}\prod_{k=7}^{n}\frac{3}{k}
    \leq\frac{2}{2^{n-6}}\\
  &\leq\frac{2}{n-5}\leq\frac{2}{N-5}<\epsilon.
\end{aligned}
\]
Here we used $2^m\geq m+1$ for $m\geq0$.
Thus $3^n/n!\to0$.
\end{proof}
\end{frame}

\begin{frame}{Problem 17}
Prove directly from the definition that
\[
  \frac{n^2}{2^n}\longrightarrow0.
\]
\textbf{Hint:} The binomial theorem gives
\[
  2^n=(1+1)^n\geq\binom{n}{3}
  =\frac{n(n-1)(n-2)}{6}\qquad(n\geq3).
\]
For $n\geq4$, compare this lower bound with a constant
multiple of $n^3$.
\end{frame}

\begin{frame}{Solution 17}
\begin{proof}
Let $\epsilon>0$. Choose
$N=\max\{4,\lfloor24/\epsilon\rfloor+1\}$.
\medskip

For $n\geq N$, we have $n-1\geq n/2$ and $n-2\geq n/2$.
The binomial theorem therefore gives
\[
  2^n\geq\binom{n}{3}
  =\frac{n(n-1)(n-2)}{6}\geq\frac{n^3}{24}.
\]
Consequently, for every $n\geq N$,
\[
  \left\lvert\frac{n^2}{2^n}-0\right\rvert
  \leq\frac{24}{n}\leq\frac{24}{N}<\epsilon.
\]
Thus $n^2/2^n\to0$.
\end{proof}
\end{frame}
